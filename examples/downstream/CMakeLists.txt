cmake_minimum_required(VERSION 3.28)
project(kanariv_downstream_examples LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(kanariv CONFIG REQUIRED)

function(add_kanariv_example name source)
  add_executable(${name} ${source})
  target_link_libraries(${name} PRIVATE ${ARGN})
endfunction()

add_kanariv_example(example_types types.cpp kanariv::types)
add_kanariv_example(example_contracts contracts.cpp kanariv::contracts)
add_kanariv_example(example_utils utils.cpp kanariv::utils)
add_kanariv_example(example_config config.cpp kanariv::config)
add_kanariv_example(example_media media.cpp kanariv::media)
add_kanariv_example(example_player player.cpp kanariv::player)
add_kanariv_example(example_database database.cpp kanariv::db)
add_kanariv_example(example_engine engine.cpp kanariv::engine kanariv::db kanariv::player)
add_kanariv_example(example_ipc ipc.cpp kanariv::ipc)
add_kanariv_example(example_client client.cpp kanariv::client)
add_kanariv_example(example_service service.cpp kanariv::service)
add_kanariv_example(example_app app.cpp kanariv::app)
add_kanariv_example(example_library library.cpp kanariv::engine kanariv::db kanariv::utils)
add_kanariv_example(example_fingerprint fingerprint.cpp kanariv::media kanariv::utils)
add_kanariv_example(example_config_backup config_backup.cpp kanariv::config kanariv::utils)
add_kanariv_example(example_mini_player mini_player.cpp kanariv::player kanariv::utils)
